cmake_minimum_required(VERSION 3.25)
project(tube360_standalone C CXX)
set(CMAKE_CXX_STANDARD 20)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_subdirectory("${root}/client/src/yt" yt)
add_executable(yt-check probe.cpp)
target_link_libraries(yt-check PRIVATE tube360-yt)
if(XSDK)
  target_compile_definitions(yt-check PRIVATE YT_XBOX)
  xsdk_add_xex(yt-check TITLE_ID 0x58544637 NAME "Tube360 standalone check")
else()
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(yt-test test.cpp)
  target_link_libraries(yt-test PRIVATE tube360-yt)
  add_executable(yt-account-test ../account/test.cpp)
  target_include_directories(yt-account-test PRIVATE "${root}/client/third_party")
  target_link_libraries(yt-account-test PRIVATE tube360-yt)
  add_test(NAME console-account COMMAND yt-account-test "${CMAKE_CURRENT_BINARY_DIR}")
  add_test(NAME standalone COMMAND yt-test "${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
  add_test(NAME reject-untrusted COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tls_test.py" "$<TARGET_FILE:yt-test>")
  set_tests_properties(reject-untrusted PROPERTIES TIMEOUT 20)
endif()
if(NOT XSDK)
  add_executable(yt-update-test ../../tests/update/contracts.cpp)
  target_link_libraries(yt-update-test PRIVATE tube360-yt)
  add_executable(yt-launcher-test ../../tests/update/launcher.cpp "${root}/client/src/update/launcher.cpp")
  target_link_libraries(yt-launcher-test PRIVATE tube360-update-core)
  add_test(NAME signed-updates COMMAND "${Python3_EXECUTABLE}" "${root}/tests/update/test.py" "$<TARGET_FILE:yt-update-test>" "$<TARGET_FILE:yt-launcher-test>")
  add_test(NAME launcher-platform COMMAND "${Python3_EXECUTABLE}" "${root}/tests/update/launcher_platform.py" "${CMAKE_CURRENT_BINARY_DIR}")
  set_tests_properties(signed-updates launcher-platform PROPERTIES TIMEOUT 30)
endif()
