cmake_minimum_required(VERSION 3.20)
project(tube360_local_library C CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
add_subdirectory(../catalog catalog)
add_executable(local-library-test test.cpp)
target_include_directories(local-library-test PRIVATE ../../client/src ../../client/third_party)
target_link_options(local-library-test PRIVATE -Wl,--wrap=write -Wl,--wrap=fsync -Wl,--wrap=rename)
add_test(NAME local-library COMMAND local-library-test "${CMAKE_CURRENT_BINARY_DIR}/storage")
add_executable(channel-art-test channel-test.cpp)
target_include_directories(channel-art-test PRIVATE ../../client/src ../../client/third_party)
target_compile_definitions(channel-art-test PRIVATE PICOJSON_USE_LOCALE=0)
target_link_libraries(channel-art-test PRIVATE tube360-yt)
add_test(NAME channel-art COMMAND channel-art-test "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(SDL3 REQUIRED CONFIG)
add_executable(local-writer-test writer-test.cpp)
target_include_directories(local-writer-test PRIVATE ../../client/src ../../client/third_party)
target_link_libraries(local-writer-test PRIVATE SDL3::SDL3)
add_test(NAME local-writer COMMAND local-writer-test "${CMAKE_CURRENT_BINARY_DIR}/writer")
add_executable(local-writer-quit-test writer-quit-test.cpp)
target_include_directories(local-writer-quit-test PRIVATE ../../client/src ../../client/third_party)
target_link_libraries(local-writer-quit-test PRIVATE SDL3::SDL3)
add_test(NAME local-writer-quit COMMAND local-writer-quit-test "${CMAKE_CURRENT_BINARY_DIR}/quit")
