cmake_minimum_required(VERSION 3.25)
project(sponsorblock_xbox_tls C CXX)
set(root "${CMAKE_CURRENT_LIST_DIR}/../../..")
set(TUBE360_TLS_XBOX_CONFIG ON)
set(MBEDTLS_SOURCE "${root}/.work/src/mbedtls-3.6.8")
set(CURL_SOURCE "${root}/.work/src/curl-xenon-a6b70bd06820e1c625f47fe0e592e08c5daaa6b4")
add_subdirectory("${root}/client/src/yt" yt)
add_executable(sponsorblock-transport-probe ../tls_probe.cpp)
target_link_libraries(sponsorblock-transport-probe PRIVATE tube360-yt)
if(XSDK)
  xsdk_add_xex(sponsorblock-transport-probe TITLE_ID 0x58544634 NAME "Tube360 SponsorBlock TLS probe")
else()
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(sponsorblock-diagnostic-test ../test.cpp)
  target_link_libraries(sponsorblock-diagnostic-test PRIVATE tube360-yt)
  add_test(NAME sponsorblock-diagnostics COMMAND sponsorblock-diagnostic-test "${root}/bench/sponsorblock/fixtures/public-prefix.json")
  add_test(NAME tls-errors COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/errors.py" "$<TARGET_FILE:sponsorblock-transport-probe>")
  set_tests_properties(tls-errors PROPERTIES TIMEOUT 30)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(SDL3 REQUIRED)
    add_executable(sponsorblock-priority-test ../priority.cpp)
    target_include_directories(sponsorblock-priority-test PRIVATE "${root}/client/src" "${root}/bench/sponsorblock/scheduler")
    target_compile_definitions(sponsorblock-priority-test PRIVATE __XSDK__=1)
    target_link_libraries(sponsorblock-priority-test PRIVATE tube360-yt SDL3::SDL3)
    add_test(NAME scheduler-control COMMAND sponsorblock-priority-test control "${root}/bench/sponsorblock/fixtures/public-prefix.json")
    add_test(NAME scheduler-boost COMMAND sponsorblock-priority-test boost "${root}/bench/sponsorblock/fixtures/public-prefix.json")
    set_tests_properties(scheduler-control scheduler-boost PROPERTIES TIMEOUT 15)
  endif()
endif()
