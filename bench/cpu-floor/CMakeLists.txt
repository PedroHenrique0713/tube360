cmake_minimum_required(VERSION 3.25)
project(cpu_floor_bench C CXX)
if(NOT FFMPEG_PREFIX)
  message(FATAL_ERROR "Use build.sh: this bench requires the private instrumented FFmpeg 7.1.5")
endif()
add_executable(cpu-floor main.cpp)
target_compile_features(cpu-floor PRIVATE cxx_std_17)
target_compile_options(cpu-floor PRIVATE -O2 -Wall -Wextra -Werror)
target_include_directories(cpu-floor PRIVATE "${FFMPEG_PREFIX}/include")
foreach(lib avformat avcodec swresample avutil)
  target_link_libraries(cpu-floor PRIVATE "${FFMPEG_PREFIX}/lib/lib${lib}.a")
endforeach()
if(XSDK)
  target_sources(cpu-floor PRIVATE ../../client/x360/ffmpeg-libc.c)
  target_link_options(cpu-floor PRIVATE -Wl,--wrap=pthread_create)
  xsdk_add_xex(cpu-floor TITLE_ID 0x58544630 NAME "Tube360 F30 CPU floor")
else()
  find_package(Threads REQUIRED)
  target_link_libraries(cpu-floor PRIVATE Threads::Threads m)
endif()
