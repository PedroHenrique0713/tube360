cmake_minimum_required(VERSION 3.25)
project(tube360_nsig C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 11)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(qjs "${root}/client/third_party/quickjs-ng")
add_library(quickjs-ng STATIC "${qjs}/quickjs.c" "${qjs}/libregexp.c" "${qjs}/libunicode.c" "${qjs}/dtoa.c")
target_compile_definitions(quickjs-ng PRIVATE _GNU_SOURCE QUICKJS_NG_BUILD)
target_include_directories(quickjs-ng PUBLIC "${qjs}")
set(NSIG_QJS_OPT "-O2" CACHE STRING "Optimisation flag for QuickJS-ng (-O2 or -Os)")
target_compile_options(quickjs-ng PRIVATE ${NSIG_QJS_OPT} -w)
if(XSDK)
  # Sem Atomics.wait/SharedArrayBuffer: o newlib do xsdk não traz pthread_condattr_setclock e o player não usa threads de JS.
  target_compile_definitions(quickjs-ng PRIVATE __STDC_NO_ATOMICS__=1)
else()
  target_link_libraries(quickjs-ng PUBLIC m)
endif()
add_library(tube360-nsig STATIC "${root}/client/src/yt/nsig.cpp")
target_include_directories(tube360-nsig PUBLIC "${root}/client/src")
target_link_libraries(tube360-nsig PUBLIC quickjs-ng)
add_executable(nsig-bench bench.cpp)
target_link_libraries(nsig-bench PRIVATE tube360-nsig)
# Linha de base do tamanho do XEX: o mesmo executável sem o motor.
add_executable(nsig-empty empty.cpp)
if(XSDK)
  target_compile_definitions(nsig-bench PRIVATE YT_XBOX)
  target_compile_definitions(nsig-empty PRIVATE YT_XBOX)
  xsdk_add_xex(nsig-bench TITLE_ID 0x58544638 NAME "Tube360 nsig bench")
  xsdk_add_xex(nsig-empty TITLE_ID 0x58544639 NAME "Tube360 nsig empty")
endif()
