cmake_minimum_required(VERSION 3.25)
project(ytspike C)

if(COMMAND xsdk_add_xex)
  add_executable(ytspike main.c)
  set_target_properties(ytspike PROPERTIES C_STANDARD 17 C_EXTENSIONS ON)
  target_include_directories(ytspike PRIVATE vendor)
  target_compile_options(ytspike PRIVATE -O2 -Wall -Wextra)
  xsdk_add_xex(ytspike)
endif()

option(TUBE360_QUALITY_BENCH "Build the four-thread player benchmark on Xbox (requires SDL-xenon)" OFF)
if(TUBE360_QUALITY_BENCH OR NOT COMMAND xsdk_add_xex)
  enable_language(CXX)
  find_package(SDL3 REQUIRED CONFIG)
  add_executable(decode-quality quality.cpp)
  target_compile_features(decode-quality PRIVATE cxx_std_20)
  target_compile_options(decode-quality PRIVATE -O2 -Wall)
  target_include_directories(decode-quality PRIVATE ../../client/src ../../client/third_party)
  if(COMMAND xsdk_add_xex)
    target_link_libraries(decode-quality PRIVATE SDL3::SDL3-static)
    xsdk_add_xex(decode-quality)
  else()
    target_link_libraries(decode-quality PRIVATE SDL3::SDL3)
  endif()
endif()
