cmake_minimum_required(VERSION 3.25)
project(f33_cpu C CXX)
if(NOT FFMPEG_PREFIX OR NOT FFMPEG_SOURCE OR NOT FFMPEG_BUILD)
  message(FATAL_ERROR "Use build.sh")
endif()
add_executable(f33-cpu main.cpp)
target_compile_features(f33-cpu PRIVATE cxx_std_17)
target_compile_options(f33-cpu PRIVATE -O2 -Wall -Wextra -Werror)
target_include_directories(f33-cpu PRIVATE "${FFMPEG_PREFIX}/include")
foreach(lib avformat avcodec swresample avutil)
  target_link_libraries(f33-cpu PRIVATE "${FFMPEG_PREFIX}/lib/lib${lib}.a")
endforeach()
if(XSDK)
  target_sources(f33-cpu PRIVATE ../../client/x360/ffmpeg-libc.c)
  target_link_options(f33-cpu PRIVATE -Wl,--wrap=pthread_create)
  xsdk_add_xex(f33-cpu TITLE_ID 0x58544633 NAME "Tube360 F33 CPU")
  add_executable(f33-kernels kernels.c)
  target_include_directories(f33-kernels PRIVATE "${FFMPEG_BUILD}" "${FFMPEG_SOURCE}" ../../tools/ffmpeg-x360)
  target_compile_options(f33-kernels PRIVATE -O3 -maltivec -mabi=altivec -fno-tree-vectorize)
  xsdk_add_xex(f33-kernels TITLE_ID 0x58544633 NAME "Tube360 F33 kernels")
else()
  find_package(Threads REQUIRED)
  target_link_libraries(f33-cpu PRIVATE Threads::Threads m)
endif()
