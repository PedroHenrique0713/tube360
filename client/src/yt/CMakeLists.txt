set(root "${CMAKE_CURRENT_LIST_DIR}/../../..")
set(ENABLE_TESTING OFF CACHE BOOL "")
set(ENABLE_PROGRAMS OFF CACHE BOOL "")
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "")
if(XSDK)
  add_subdirectory("${root}/tools/mbedtls-x360" tls)
else()
  set(MBEDTLS_USER_CONFIG_FILE "${CMAKE_CURRENT_LIST_DIR}/tls-threading.h")
  add_subdirectory("${MBEDTLS_SOURCE}" tls)
endif()
set(MBEDTLS_INCLUDE_DIR "${MBEDTLS_SOURCE}/include" CACHE PATH "")
set(MBEDTLS_LIBRARY mbedtls CACHE STRING "")
set(MBEDX509_LIBRARY mbedx509 CACHE STRING "")
set(MBEDCRYPTO_LIBRARY mbedcrypto CACHE STRING "")
include("${CURL_SOURCE}/CMake/xsdk.cmake")
set(CURL_DISABLE_INSTALL ON CACHE BOOL "")
# A sessão de visitante do YouTube vive em cookies (YSC, VISITOR_INFO1_LIVE, __Secure-YNID)
set(CURL_DISABLE_COOKIES OFF CACHE BOOL "" FORCE)
add_subdirectory("${CURL_SOURCE}" curl)
file(READ "${root}/client/src/yt/gts.pem" roots)
configure_file("${root}/bench/standalone/roots.hpp.in" roots.hpp @ONLY)
add_library(tube360-yt STATIC "${root}/client/src/yt/https.cpp" "${root}/client/src/yt/client.cpp"
  "${root}/client/src/yt/console_account.cpp" "${root}/client/src/yt/account_transport.cpp" "${root}/client/src/yt/cast.cpp")
target_compile_features(tube360-yt PUBLIC cxx_std_17)
target_include_directories(tube360-yt PUBLIC "${root}/client/src" PRIVATE "${root}/client/third_party" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(tube360-yt PRIVATE PICOJSON_USE_LOCALE=0)
target_link_libraries(tube360-yt PRIVATE CURL::libcurl)

file(READ "${root}/client/src/update/rules-default.json" default_rules)
configure_file("${root}/client/src/update/rules-default.json" "${CMAKE_CURRENT_BINARY_DIR}/rules-default.json" COPYONLY)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/rules-default.hpp" "#pragma once\nnamespace updates { inline constexpr const char* defaultRules = R\"RULES(${default_rules})RULES\"; }\n")
file(READ "${root}/client/src/update/roots.pem" update_roots)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/update-roots.hpp" "static constexpr char updateRoots[] = R\"PEM(${update_roots})PEM\";\n")
add_library(tube360-boot STATIC "${root}/client/src/boot_log.c" "${root}/client/src/boot_early.c" "${root}/client/src/boot_platform.cpp" "${root}/client/src/video_policy.c" "${root}/client/src/board_sim.c")
set(TUBE360_NATIVE_BOARD "unconfirmed" CACHE STRING "Exact hardware family qualified on the owner's console")
set(native_families xenon zephyr falcon jasper trinity corona winchester)
list(FIND native_families "${TUBE360_NATIVE_BOARD}" native_family)
if(native_family EQUAL -1 AND NOT TUBE360_NATIVE_BOARD STREQUAL "unconfirmed")
  message(FATAL_ERROR "Unknown TUBE360_NATIVE_BOARD: ${TUBE360_NATIVE_BOARD}")
endif()
target_compile_definitions(tube360-boot PRIVATE TUBE360_NATIVE_BOARD_ID=${native_family})
target_include_directories(tube360-boot PUBLIC "${root}/client/src")
target_compile_features(tube360-boot PUBLIC cxx_std_17)

add_library(tube360-update-core STATIC "${root}/client/src/update/common.cpp" "${root}/client/src/update/app.cpp"
  "${root}/client/third_party/monocypher/monocypher.c" "${root}/client/third_party/monocypher/monocypher-ed25519.c")
target_compile_features(tube360-update-core PUBLIC cxx_std_17)
target_compile_definitions(tube360-update-core PUBLIC PICOJSON_USE_LOCALE=0)
target_include_directories(tube360-update-core PUBLIC "${root}/client/src" "${root}/client/third_party" PRIVATE "${root}/client/third_party/monocypher")
target_link_libraries(tube360-update-core PUBLIC mbedcrypto tube360-boot)
target_sources(tube360-yt PRIVATE "${root}/client/src/update/rules.cpp")
target_link_libraries(tube360-yt PUBLIC tube360-update-core)
set(TUBE360_BUILD_ID 1 CACHE STRING "Monotonic signed app build")
set(TUBE360_VERSION "0.3.0-dev" CACHE STRING "Signed app version")
target_compile_definitions(tube360-update-core PUBLIC TUBE360_BUILD_ID=${TUBE360_BUILD_ID} TUBE360_VERSION="${TUBE360_VERSION}")
